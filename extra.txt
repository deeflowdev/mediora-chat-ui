    <script>
        const app = {
            chats: JSON.parse(localStorage.getItem('mediora_slate')) || [],
            currId: null,

            init() {
                if (this.chats.length === 0) this.newChat();
                else this.loadChat(this.chats[0].id);
                
                document.getElementById('user-input').oninput = function() {
                    this.style.height = 'auto';
                    this.style.height = this.scrollHeight + 'px';
                };
            },

            newChat() {
                const id = Date.now();
                this.chats.unshift({ id, title: "New Session", msgs: [] });
                this.loadChat(id);
            },

            loadChat(id) {
                this.currId = id;
                this.render();
            },

            send() {
                const input = document.getElementById('user-input');
                if (!input.value.trim()) return;

                const chat = this.chats.find(c => c.id === this.currId);
                if (chat.msgs.length === 0) chat.title = input.value.slice(0, 20) + "...";

                chat.msgs.push({ role: 'user', text: input.value });
                input.value = '';
                input.style.height = 'auto';

                // Bot Response
                setTimeout(() => {
                    chat.msgs.push({ role: 'bot', text: "Analyzing your request within the slate framework. How else can I help?" });
                    this.render();
                    localStorage.setItem('mediora_slate', JSON.stringify(this.chats));
                }, 600);
                
                this.render();
            },

            render() {
                const list = document.getElementById('chat-list');
                list.innerHTML = this.chats.map(c => `
                    <div class="chat-item ${c.id === this.currId ? 'active' : ''}" onclick="app.loadChat(${c.id})">
                        <i class="ph ph-chat-teardrop"></i> ${c.title}
                    </div>
                `).join('');

                const chat = this.chats.find(c => c.id === this.currId);
                const container = document.getElementById('chat-container');
                container.innerHTML = chat.msgs.map(m => `
                    <div class="message ${m.role}">
                        <div class="bubble">${m.text}</div>
                        <div class="meta"><span>Just now</span></div>
                    </div>
                `).join('');
                container.scrollTop = container.scrollHeight;
            }
        };
        app.init();
    </script>
